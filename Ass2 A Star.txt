#include <iostream>
using namespace std;

class Graph
{
    struct node
    {
        int data;
        int weight;
        node *next;
    };

public:
    node *head[20];
    int Heuristic[20];
    int no, fn, gn;

    Graph()
    {
        no = 0;

        for (int i = 0; i < 20; i++)
        {
            head[i] = NULL;
            Heuristic[i] = 0;
        }
    }

    void edge_v1_v2(int, int, int);
    void display();
    void Astar(int, int);
};

// Insert Edge
void Graph::edge_v1_v2(int v1, int v2, int w)
{
    node *n, *f, *h;

    h = new node;
    h->data = v1;
    h->next = NULL;

    n = new node;
    n->data = v2;
    n->weight = w;
    n->next = NULL;

    f = head[v1];

    if (f == NULL)
    {
        h->next = n;
        head[v1] = h;
    }
    else
    {
        while (f->next)
            f = f->next;

        f->next = n;
    }
}

// Display Graph
void Graph::display()
{
    node *f;

    for (int i = 0; i < no; i++)
    {
        cout << "\n"
             << i << " -> ";

        f = head[i];

        if (f)
            f = f->next;

        while (f)
        {
            cout << f->data << "(" << f->weight << ") ";

            f = f->next;
        }
    }
}

// A* Algorithm
void Graph::Astar(int src, int goal)
{
    node *f;

    gn = 0;

    cout << "\nShortest Path is : ";
    cout << src << "\t";

    while (src != goal)
    {
        f = head[src];
        f = f->next;

        fn = 999;

        int nextNode = src;
        int minCost = 999;

        while (f)
        {
            int cost = gn + f->weight + Heuristic[f->data];

            if (cost < minCost)
            {
                minCost = cost;
                nextNode = f->data;
                fn = cost;
            }

            f = f->next;
        }

        gn = fn - Heuristic[nextNode];
        src = nextNode;

        cout << src << "\t";
    }
}

// Main Function
int main()
{
    Graph g;

    g.no = 5;

    // Heuristic values
    g.Heuristic[0] = 7;
    g.Heuristic[1] = 6;
    g.Heuristic[2] = 2;
    g.Heuristic[3] = 1;
    g.Heuristic[4] = 0;

    // Edges
    g.edge_v1_v2(0, 1, 2);
    g.edge_v1_v2(0, 2, 4);
    g.edge_v1_v2(1, 3, 3);
    g.edge_v1_v2(2, 4, 1);
    g.edge_v1_v2(3, 4, 2);

    cout << "\nGraph:";
    g.display();

    cout << "\n";
    g.Astar(0, 4);

    return 0;
}
