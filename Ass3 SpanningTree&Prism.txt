#include <iostream>
#include <vector>
using namespace std;

class Graph
{
    int cost[10][10], n;

public:
    Graph(int vertices)
    {
        n = vertices;

        for(int i = 0; i < n; i++)
            for(int j = 0; j < n; j++)
                cost[i][j] = 999;
    }

    void addEdge(int u, int v, int w)
    {
        cost[u][v] = w;
        cost[v][u] = w;
    }

    void display()
    {
        cout << "\nGraph:\n";

        for(int i = 0; i < n; i++)
        {
            cout << i << " -> ";

            for(int j = 0; j < n; j++)
            {
                if(cost[i][j] != 999)
                    cout << j << "(" << cost[i][j] << ") ";
            }

            cout << endl;
        }
    }

    void prims()
    {
        int visited[10] = {0};

        visited[0] = 1;

        int edges = 0;
        int mincost = 0;

        cout << "\nMinimum Spanning Tree:\n";

        while(edges < n - 1)
        {
            int min = 999, a = -1, b = -1;

            for(int i = 0; i < n; i++)
            {
                if(visited[i])
                {
                    for(int j = 0; j < n; j++)
                    {
                        if(!visited[j] && cost[i][j] < min)
                        {
                            min = cost[i][j];
                            a = i;
                            b = j;
                        }
                    }
                }
            }

            cout << a << " -> " << b << " = " << min << endl;

            visited[b] = 1;
            mincost += min;
            edges++;
        }

        cout << "\nMinimum Cost = " << mincost << endl;
    }
};

int main()
{
    Graph g(5);

    g.addEdge(0, 1, 2);
    g.addEdge(0, 3, 6);
    g.addEdge(1, 2, 3);
    g.addEdge(1, 3, 8);
    g.addEdge(1, 4, 5);
    g.addEdge(2, 4, 7);
    g.addEdge(3, 4, 9);

    g.display();

    g.prims();

    return 0;
}
