#include <iostream>
#include <cmath>
using namespace std;

int x[20];

// Check whether queen can be placed
int place(int k, int i)
{
    for(int j = 1; j <= k - 1; j++)
    {
        // Same column or diagonal
        if(x[j] == i || abs(x[j] - i) == abs(j - k))
            return 0;
    }

    return 1;
}

// N-Queens using Backtracking
void NQueens(int k, int n)
{
    for(int i = 1; i <= n; i++)
    {
        if(place(k, i))
        {
            x[k] = i;

            if(k == n)
            {
                cout << "\nSolution:\n";

                for (int row = 1; row <= n; row++)
                {
                    for (int col = 1; col <= n; col++)
                    {
                        if (x[row] == col)
                            cout << " Q ";
                        else
                            cout << " . ";
                    }

                    cout << endl;
                }
            }
            else
            {
                NQueens(k + 1, n);
            }
        }
    }
}

// Main Function
int main()
{
    int n;

    cout << "Enter number of queens: ";
    cin >> n;

    NQueens(1, n);

    return 0;
}
